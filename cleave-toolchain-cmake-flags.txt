# The exact CMake configuration cleave's own OpenBLAS toolchain needs --
# single source of truth for this repo's own build-release workflow, so the
# Windows and Linux jobs (and any local rebuild) never drift from each
# other by copy-paste.
#
# One `-D<name>=<value>` per line, `#`-prefixed comments and blank lines
# ignored by the workflow's own parsing.

CMAKE_BUILD_TYPE=Release
# No Fortran toolchain anywhere in this build, on either platform --
# `cleave` only ever needs the BLAS entry points (`sgemm`/`dgemm`/...), never
# LAPACK's own factorizations. `NOFORTRAN=1` skips LAPACK/Fortran entirely
# (the well-established, real escape hatch OpenBLAS's own WASM/Emscripten
# and embedded/Android builds already use for the identical reason) --
# confirmed directly that this does not touch the optimized BLAS Level 3
# kernels at all (those are hand-written C/assembly/intrinsics in
# `kernel/`, never Fortran).
NOFORTRAN=1
# Every CPU tier built into one binary, real runtime dispatch by CPUID
# feature detection (never a vendor-string check) -- confirmed directly,
# not just documented, that this correctly routes a real AMD Zen 5 chip
# with AVX-512 to the SKYLAKEX/COOPERLAKE kernel tier, not the AVX2-only
# ZEN one (`cpuid_x86.c`'s own `case 11: // Zen5` branch). The whole point
# of a redistributable binary -- one artifact that's fast on whatever CPU
# it actually runs on, AMD or Intel, not tuned for the build machine alone.
DYNAMIC_ARCH=ON
# `USE_OPENMP=1` used to live here instead, on the theory that sharing
# `cleave`'s own OpenMP runtime (`--convert-scf-to-openmp`/`--convert-
# openmp-to-llvm`) beats racing it with a separate pthread pool, plus a
# claim that `openblas_set_num_threads(1)` takes a real, separate serial
# code path internally, no OpenMP team involved even at n=1 -- reverted
# after directly, empirically contradicting both: VTune on a real
# `cleave`-generated binary (`examples/mnist-interop`, single-threaded,
# `openblas_set_num_threads(1)` called before every BLAS entry) still
# showed real time inside `libomp140_x86_64.dll`, and cleave's own
# parallelization strategy lives one level up, entirely in cleave's own
# generated code (`--openmp`) -- BLAS is never asked for more than one
# thread from cleave today, so there is no thread pool left to "share" in
# the first place. `USE_THREAD=0`: a genuinely serial build, no pthread,
# no OpenMP, no threading runtime at all to initialize or race against --
# `openblas_set_num_threads` still links (a real no-op stub in a serial
# build), so `cleave-rt`'s own unconditional call to it needs no change.
USE_THREAD=0
BUILD_SHARED_LIBS=ON
